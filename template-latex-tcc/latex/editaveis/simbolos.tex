\begin{simbolos}
  \item[$ H_0 $] Hipótese nula
  \item[$ H_1 $] Hipótese alternativa
  \item[$ A_i $] Quantidade de concordância do tratamento $i$
  \item[$ i $] Índice do tratamento, onde $i \in \{1,2,3,4,5,6,7\}$
  \item[$ Q $] Estatística do teste de Cochran's Q
  \item[$ W $] Coeficiente de Kendall
  \item[$ n $] Número de blocos experimentais
  \item[$ k $] Número de tratamentos avaliados
  \item[$ \in $] Pertence (relação de pertinência)
  \item[$ \exists $] Existe (quantificador existencial)
  \item[$ \mid $] Tal que (condição)
  \item[$ \neq $] Diferente (desigualdade)
  \item[$ = $] Igual (igualdade)
  \item[$\alpha$] Nível de significância (probabilidade de erro do Tipo I)

\end{simbolos}
